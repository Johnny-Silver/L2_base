function ipv4Parser(ip, mask){
  const ipA = ip.split('.').map(Number);
  const maskA = mask.split('.').map(Number);
  const arr_1 = [];
  const arr_2 =[];
  
  for(let i = 0; i<4; i++){
    arr_1.push(ipA[i] & maskA[i]);
    arr_2.push(ipA[i] & (~maskA[i] & 255));
  }
  return [arr_1.join('.'), arr_2.join('.')];
}

function whatCentury(year)
{
  const a = Math.ceil(Number(year)/100);
  if(a %100 >= 11 && a %100 <= 13){
    return a+"th";
  }
  
  switch (a%10){
      case 1: return a+"st";
      case 2: return a+"nd";
      case 3: return a+"rd";
      default: return a+"th";
  }
}

function findMissing(list) {  
  const a = (list[list.length-1] - list[0]) /list.length;
  
  for(let i = 1; i<list.length; i++){
    if(list[i]-list[i-1]!==a){
      return list[i-1] + a;
    }
  }
}

function primeFactors(n){
    let result = "";
  for(let d = 2; d*d<=n; d++){
    let c = 0;
    while(n%d===0){
      c++;
      n/=d;
    }
    
    if(c>0){
      result += c ===1 ? `(${d})` : `(${d}**${c})`;
    }
  }
  if(n>1){
    result+=`(${n})`;
  }
  return result;
}

function toWeirdCase(string){
  return string.split(' ').map(a=>{
    let result = '';
    for(let i =0; i<a.length; i++){
      result+=i%2===0 ? a[i].toUpperCase() : a[i].toLowerCase();
    }
    return result;
  }).join(' ');
}

function bingo(ticket, win){
  let i = 0;
  for(const [str, n] of ticket){
    if([...str].some(ch=>ch.charCodeAt(0)=== n)){
      i++;
    }
  }
  return i >= win ? 'Winner!' : 'Loser!';
}

  let c = 0;
  for(let i =0; i<arr.length;  i++){
    if(arr[i]>arr[(i+1)%arr.length]){
      c++;
    }
  }
  return c<=1;
}

function howManyTimes(time1, time2) {
  let start = Date.parse(time1.replace(' ', 'T') + 'Z');
  let end = Date.parse(time2.replace(' ', 'T') + 'Z');
  let totalStrikes = 0;
  let current = new Date(start);
  current.setUTCMinutes(0, 0, 0);
  let currentTime = current.getTime();
   if (currentTime > start) {
    currentTime -= 60 * 60 * 1000;
  }
  
  while (currentTime <= end) {
    let date = new Date(currentTime);
    let minutes = date.getUTCMinutes();
    let hours = date.getUTCHours();
    
    let strikesAtThisMoment = 0;
    if (minutes === 0) {
      let h12 = hours % 12;
      strikesAtThisMoment = (h12 === 0 ? 12 : h12);
    } else if (minutes === 30) {
      strikesAtThisMoment = 1;
    }
    
    for (let s = 0; s < strikesAtThisMoment; s++) {
      let strikeTime = currentTime + (s * 1000);
      if (strikeTime >= start && strikeTime < end) {
        totalStrikes++;
      }
    }
    
    currentTime += 30 * 60 * 1000;
  }
  
  return totalStrikes;
}

function cache(func) {
  const cached = {};
  
  return function (...args){
    const key = JSON.stringify(args);
    
    if(key in cached){
      return cached[key];
    }
    const result = func(...args);
    cached[key]=result;
    
    return result;
  };
}

function createPhoneNumber(numbers){
  let format = "(xxx) xxx-xxxx";
  
  for(let n of numbers){
    format = format.replace('x', n);
  }
  return format;
}
